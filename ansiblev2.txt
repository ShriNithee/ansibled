mkdir ansible-modules
cd ansible-modules
nano application.yml
---
- name: Configure Simple Web Application
  hosts: localhost
  become: yes
  connection: local

  tasks:
    - name: Install Apache
      apt:
        name: apache2
        state: present
        update_cache: yes

    - name: Create web directory
      file:
        path: /var/www/html
        state: directory

    - name: Create HTML page
      copy:
        content: "<h1>Welcome to Ansible Application</h1>"
        dest: /var/www/html/index.html

    - name: Start Apache service
      service:
        name: apache2
        state: started
        enabled: yes

    - name: Verify application
      uri:
        url: http://localhost
        status_code: 200
-----or-----
---
- name: Install and Configure Apache
  hosts: localhost
  become: yes
  connection: local

  tasks:
    - name: Install Apache
      apt:
        name: apache2
        state: present
        update_cache: yes

    - name: Start and enable Apache
      service:
        name: apache2
        state: started
        enabled: yes

    - name: Deploy welcome HTML page
      copy:
        content: |
          <html>
          <head>
              <title>Welcome Page</title>
          </head>
          <body>
              <h1>Welcome to Apache Web Server</h1>
          </body>
          </html>
        dest: /var/www/html/index.html

    - name: Verify web application
      uri:
        url: http://localhost
        status_code: 200
ansible-playbook -i localhost, application.yml
curl http://localhost
sudo service apache2 status
